🎨 Color Guessing Game - Source Code

const COLORS_ARRAY = ["blue", "cyan", "gold", "gray", "green", "magenta", "orange", "red", "white", "yellow"];

function runGame() { 
  let guess = "";
  let correct = false;
  let numTries = 0;

  const targetIndex = Math.floor(Math.random() * COLORS_ARRAY.length);
  const target = COLORS_ARRAY[targetIndex];

  do {
    guess = prompt("I am thinking of one of these colors:\n\n" + COLORS_ARRAY.join(", ") + "\n\nWhat color am I thinking of?");
    guess = guess.toLowerCase().trim();
    numTries++;
    correct = checkGuess(guess, target);
  } while (!correct);
      
  alert("🎉 You correctly guessed " + target + "! It took you " + numTries + " tries.");
  document.body.style.background = target;
}

😊 Matching Game - Source Code

let numberOfFaces = 5;
let gameActive = true;
const leftSide = document.getElementById("leftSide");
const rightSide = document.getElementById("rightSide");

// Clear both sides using while loops (per assignment spec)
function clearFaces() {
  while (leftSide.firstChild) {
    leftSide.removeChild(leftSide.firstChild);
  }
  while (rightSide.firstChild) {
    rightSide.removeChild(rightSide.firstChild);
  }
}

// Game over handler — attached once to document.body
function gameOver() {
  if (!gameActive) return;
  gameActive = false;
  alert("Game Over!");
  document.body.removeEventListener("click", gameOver);
}

function generateFaces() {
  // Clear previous faces using while loops
  clearFaces();

  // Generate faces on the left side
  for (let i = 0; i < numberOfFaces; i++) {
    const face = document.createElement("img");
    face.src = 'images/smile.png';
    face.style.top = Math.floor(Math.random() * 400) + 'px';
    face.style.left = Math.floor(Math.random() * 400) + 'px';
    leftSide.appendChild(face);
  }

  // Clone leftSide, remove last child (extra face), append to rightSide
  const leftClone = leftSide.cloneNode(true);
  leftClone.removeChild(leftClone.lastChild);
  while (leftClone.firstChild) {
    rightSide.appendChild(leftClone.firstChild);
  }

  // Extra face click advances level
  const extraFace = leftSide.lastChild;
  extraFace.addEventListener("click", function handleCorrectClick(event) {
    if (!gameActive) return;
    event.stopPropagation();
    numberOfFaces += 5;
    generateFaces();
  });
}

// Attach game over listener once to document.body
document.body.addEventListener("click", gameOver);
generateFaces();
