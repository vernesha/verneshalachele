🌊 Wild Waves - Source Code

const WAVE_POOL_MIN_HEIGHT = 42; // in inches
const WATERSLIDE_MIN_HEIGHT = 48; // in inches
const LAZY_RIVER_MIN_HEIGHT = 36; // in inches

function buyTicket() {
  const attractionType = prompt(
    'Which attraction do you want to visit?\n\nType:\n• "wave" for Wave Pool (42" min)\n• "water" for Waterslide (48" min)\n• "river" for Lazy River (36" min)'
  );
  
  if (!attractionType) {
    return; // User cancelled
  }
  
  const userHeight = prompt("What is your height in inches?");
  
  if (!userHeight) {
    return; // User cancelled
  }
  
  const isAllowed = checkHeight(attractionType.toLowerCase(), parseInt(userHeight));
  
  console.log("User's selected attraction is " + attractionType);
  console.log("User's height is " + userHeight + " inches");
  
  if (isAllowed) {
    alert("🎉 Enjoy the " + getAttractionName(attractionType) + "!");
  } else {
    alert("❌ Sorry, you are not tall enough for this ride.\n\nPlease check the height requirements!");
  }
}

function checkHeight(attraction, height) {
  if (attraction === "wave") {
    return height >= WAVE_POOL_MIN_HEIGHT;
  } else if (attraction === "water") {
    return height >= WATERSLIDE_MIN_HEIGHT;
  } else if (attraction === "river") {
    return height >= LAZY_RIVER_MIN_HEIGHT;
  } else {
    alert("⚠️ Invalid attraction type. Please enter 'wave', 'water', or 'river'.");
    return false;
  }
}

function getAttractionName(type) {
  const names = {
    'wave': 'Wave Pool',
    'water': 'Waterslide',
    'river': 'Lazy River'
  };
  return names[type.toLowerCase()] || type;
}

🎨 Color Guessing Game - Source Code

const COLORS_ARRAY = ["blue", "cyan", "gold", "gray", "green", "magenta", "orange", "red", "white", "yellow"];

function runGame() { 
  let guess = "";
  let correct = false;
  let numTries = 0;

  // Randomly select a target color
  const targetIndex = Math.floor(Math.random() * COLORS_ARRAY.length);
  const target = COLORS_ARRAY[targetIndex];

  do {
    guess = prompt("I am thinking of one of these colors:\n\n" + COLORS_ARRAY.join(", ") + "\n\nWhat color am I thinking of?");
     
    if (guess === null) {
      alert("Game cancelled. Goodbye!");
      return;   
    }
      
    guess = guess.toLowerCase().trim();
    numTries++;

    correct = checkGuess(guess, target);
  } while (!correct);
      
  alert("🎉 You correctly guessed " + target + "! It took you " + numTries + " tries.");
  document.body.style.background = target;
}

function checkGuess(guess, target) { 
  let correct = false; 

  if (!COLORS_ARRAY.includes(guess)) {
    alert("❌ That color is not in the list.");
  } else if (guess > target) {
    alert("📉 Your guess is alphabetically higher.");
  } else if (guess < target) {
    alert("📈 Your guess is alphabetically lower.");
  } else {
    correct = true;
  }
   
  return correct;
}
